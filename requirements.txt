# Python 3.10-3.12 is required. See requirements-dev.txt and README for the

# --- Simulation engines (Tier 2 ODE domains) -------------------------------
# NOTE: do NOT `pip install tellurium`. The umbrella package pulls in
# python-libcombine and python-libnuml, which are only needed for COMBINE
# archives and numerical markup -- neither of which Terrium uses. On any
# platform without prebuilt wheels for them, that install fails at the cmake
# step. These three packages are the parts that actually do the work.
#
# libroadrunner is pinned to 2.7.0, not the newer 2.9.x line. 2.9.0/2.9.2
# dropped cp310 wheels entirely (their PyPI releases start at cp311), which
# breaks installation on Python 3.10 -- the floor this project claims to
# support. This was caught because CI failed on the 3.10 job with
# "No matching distribution found for libroadrunner==2.9.0". 2.7.0 ships
# wheels for cp39 through cp312 and is what the full test suite has actually
# been verified against.

antimony==2.14.0
